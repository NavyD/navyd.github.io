如何定义一个混合类型的参数

有一个需求是函数的参数接收命令参数数组 string[] 或 scriptblock 允许执行命令或代码块，可参考 gsudo usage 中类似 gsudo echo hello, gsudo { echo hello } 。

在 gsudo/src/gsudo.Wrappers/Invoke-gsudo.ps1 有很好的实现，但没有混合类型，在 deepseek 提示中可以使用 object[] 与 ValueFromRemainingArguments 属性收集所有未明确绑定的位置参数到 $Cmd 中，允许 Test-Func -Wait { echo hello; echo world } 和 Test-Func -Wait echo hello world。

function Test-Func {
    [CmdletBinding()]
    param(
        [Parameter(Position=0, Mandatory=$true, ValueFromRemainingArguments=$true)]
        [object[]]$Cmd,

        [Parameter()]
        [switch]$Wait
    )
}

在函数中可以使用 -is/isnot 检查 $Cmd[0] -is [scriptblock] 判断是否是脚本块。

function Test-Args {
    param($ModuleName)
    Start-Job -ScriptBlock {
        function Testecho {
            echo "test scriptblock outside args: $args"
        }
        Testecho @Using:args @Using:PSBoundParameters
    } | Receive-Job -AutoRemoveJob -Wait
}

# test scriptblock outside args: -ModuleName: hello
Test-Args hello world
# test scriptblock outside args: world -ModuleName: hello

function Import-CmdOutput {
    [CmdletBinding(PositionalBinding = $false)]
    param(
        [Parameter(Mandatory = $false)]
        [string]$ModuleName,
        [parameter(Mandatory, ValueFromRemainingArguments)]
        [object[]]$Commands
    )
    function TestArgs {
        "args: $args"
    }
    [scriptblock]$sb = { TestArgs -WaitIfNoCache @Using:Commands | Out-String | Invoke-Expression }
    if ($ModuleName) {
        New-Module -Name $ModuleName -ScriptBlock $sb | Import-Module -Global
    }
    else {
        Invoke-Command -ScriptBlock $sb
    }
}

function Import-CmdOutput {
    [CmdletBinding(PositionalBinding = $false)]
    param(
        [Parameter(Mandatory = $false)]
        [string]$ModuleName,
        [parameter(Mandatory, ValueFromRemainingArguments)]
        [object[]]$Commands
    )
    function TestArgs {
        "args: $args"
    }
    [scriptblock]$sb = { TestArgs -WaitIfNoCache @args }
    if ($ModuleName) {
        New-Module -Name $ModuleName -ScriptBlock $sb -ArgumentList $Commands | Import-Module -Global
    }
    else {
        Invoke-Command -ScriptBlock $sb -ArgumentList $Commands
    }
}

Import-CmdOutput zoxide init powershell
# args: -WaitIfNoCache zoxide init powershell
