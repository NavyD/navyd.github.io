#HotIf WinActive("ahk_exe emacs.exe")
^Space::{
    SetKeyDelay 75, 25
    ControlSend "^{Space}", , "ahk_exe emacs.exe"
}

#Requires AutoHotkey v2.0
#SingleInstance

; 键入`<C-space>`时发送`<C-space>`到当前活动窗口
^Space::{
    ; [WinTitle Parameter & Last Found Window](https://www.autohotkey.com/docs/v2/misc/WinTitle.htm#ActiveWindow)
    ; A表示当前活动窗口
    ControlSend "^{Space}",, "A"
}

对于部分程序如 windows terminal 这个无效，退出 ahk 后可使用 ENG 美式键盘在 wsl 中 showkey -a 输入 ^@ 0 0000 0x00 。在 vscode,neovide 中 ctrl+space 可正常工作，我尝试了 alacritty,wezterm,windowsterminal 即使在美式键盘下 neovim 也无法正常使用。而 windows terminal 中使用 powershell 在美式键盘下输入 c-space 正常工作，启用这段程序后 c-space 无效，可能不只是终端的问题 Control+Space not sent to program running in terminal #15939 ，与 neovim 实践：windows 上使用 <C-space> 无效 不同，这不是 windows terminal 的问题

在使用键盘监听程序测试时如 “Key-Test” - keyboard test online 在按下 ctrl+space 时会发送 ctrl+space, ctrl+space 两次，这可能会导致问题

注册表

从 如何改变 Windows 默认的输入法快捷键 Ctrl+Space 为其他值？ 中发现只需要改 HKEY_USERS\.DEFAULT\Control Panel\Input Method\Hot Keys 的 00000010 和 00000070 就可以了，修改步骤参考 CTRL-Space always toggles Chinese IME (Windows 7) ，注意 REG_BINARY 值的转换参考 How to set a binary registry value (REG_BINARY) with PowerShell? ，如 Key Modifiers 表示为 左Alt+Shift 时在注册表中显示 05 c0 00 00 转换处理 ([byte[]](0x05, 0xc0, 0x00, 0x00))

# 定位到 Ctrl+Space 对应的注册表项（通常是 00000070）
$basePath = "HKCU:\Control Panel\Input Method\Hot Keys"
# 70 表示 cht, 10 表示 chs
foreach($ime_key in ("00000070", "00000010")) {
  $keyPath = Join-Path $basePath $ime_key
  # Set-ItemProperty -Path $keyPath -Name "Key Modifiers" -Value ([byte[]](0x05, 0xc0, 0x00, 0x00))
  Set-ItemProperty -Path $keyPath -Name "Key Modifiers" -Value 0x00
  Set-ItemProperty -Path $keyPath -Name "Virtual Key" -Value 0xFF
}

注意：使用图形界面修改默认热键只需要重启应用界面即可使得 ctrl+space 生效，但直接修改注册表可能需要重新登录或重启才能生效
